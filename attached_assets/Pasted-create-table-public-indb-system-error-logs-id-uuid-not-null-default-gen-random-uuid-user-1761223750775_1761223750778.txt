create table public.indb_system_error_logs (
  id uuid not null default gen_random_uuid (),
  user_id uuid null,
  error_type text not null,
  severity text not null,
  message text not null,
  user_message text not null,
  endpoint text null,
  http_method text null,
  status_code integer null,
  metadata jsonb null,
  stack_trace text null,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  resolved_at timestamp with time zone null,
  resolved_by uuid null,
  acknowledged_at timestamp with time zone null,
  acknowledged_by uuid null,
  constraint indb_system_error_logs_pkey primary key (id),
  constraint indb_system_error_logs_user_id_fkey foreign KEY (user_id) references auth.users (id) on delete set null,
  constraint indb_system_error_logs_resolved_by_fkey foreign KEY (resolved_by) references auth.users (id),
  constraint indb_system_error_logs_acknowledged_by_fkey foreign KEY (acknowledged_by) references auth.users (id),
  constraint indb_system_error_logs_severity_check check (
    (
      severity = any (
        array[
          'LOW'::text,
          'MEDIUM'::text,
          'HIGH'::text,
          'CRITICAL'::text
        ]
      )
    )
  ),
  constraint indb_system_error_logs_http_method_check check (
    (
      http_method = any (
        array[
          'GET'::text,
          'POST'::text,
          'PUT'::text,
          'PATCH'::text,
          'DELETE'::text
        ]
      )
    )
  ),
  constraint indb_system_error_logs_error_type_check check (
    (
      error_type = any (
        array[
          'AUTHENTICATION'::text,
          'AUTHORIZATION'::text,
          'VALIDATION'::text,
          'DATABASE'::text,
          'EXTERNAL_API'::text,
          'ENCRYPTION'::text,
          'RATE_LIMITING'::text,
          'SYSTEM'::text,
          'NETWORK'::text,
          'BUSINESS_LOGIC'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_error_logs_unresolved on public.indb_system_error_logs using btree (created_at desc) TABLESPACE pg_default
where
  (resolved_at is null);

create index IF not exists idx_error_logs_severity_unresolved on public.indb_system_error_logs using btree (severity, created_at desc) TABLESPACE pg_default
where
  (resolved_at is null);

create index IF not exists idx_error_logs_endpoint on public.indb_system_error_logs using btree (endpoint) TABLESPACE pg_default
where
  (endpoint is not null);

create index IF not exists idx_indb_system_error_logs_user_id on public.indb_system_error_logs using btree (user_id) TABLESPACE pg_default;

create index IF not exists idx_indb_system_error_logs_error_type on public.indb_system_error_logs using btree (error_type) TABLESPACE pg_default;

create index IF not exists idx_indb_system_error_logs_severity on public.indb_system_error_logs using btree (severity) TABLESPACE pg_default;

create index IF not exists idx_indb_system_error_logs_created_at on public.indb_system_error_logs using btree (created_at desc) TABLESPACE pg_default;

create index IF not exists idx_indb_system_error_logs_endpoint on public.indb_system_error_logs using btree (endpoint) TABLESPACE pg_default;

create index IF not exists idx_indb_system_error_logs_status_code on public.indb_system_error_logs using btree (status_code) TABLESPACE pg_default;

create index IF not exists idx_indb_system_error_logs_user_severity_date on public.indb_system_error_logs using btree (user_id, severity, created_at desc) TABLESPACE pg_default;

create index IF not exists idx_error_logs_severity on public.indb_system_error_logs using btree (severity) TABLESPACE pg_default;

create index IF not exists idx_error_logs_error_type on public.indb_system_error_logs using btree (error_type) TABLESPACE pg_default;

create index IF not exists idx_error_logs_created_at on public.indb_system_error_logs using btree (created_at desc) TABLESPACE pg_default;

create index IF not exists idx_error_logs_user_id on public.indb_system_error_logs using btree (user_id) TABLESPACE pg_default;

create index IF not exists idx_error_logs_resolved on public.indb_system_error_logs using btree (resolved_at) TABLESPACE pg_default
where
  (resolved_at is null);
